// 1. Reverse linked list
Node* reverse(Node* head) {
    Node *prev = NULL, *next;
    while (head) {
        next = head->next;
        head->next = prev;
        prev = head;
        head = next;
    }
    return prev;
}

// 4. Missing number (1 to n)
int missing(int a[], int n) {
    int xor = 0;
    for(int i=1;i<=n;i++) xor ^= i ^ a[i-1];
    return xor;
}

// 7. Power(x,n)
double pow(double x, int n) {
    if(n<0) return 1/pow(x,-n);
    double res=1;
    while(n){
        if(n&1) res *= x;
        x *= x; n>>=1;
    }
    return res;
}
